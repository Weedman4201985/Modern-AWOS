sub code_to_data {
	my $cref = shift;
	my $code = $cref;
	my $chance = 100;
	my $color = undef;
	if (ref($cref) eq "ARRAY") {
		($code, $chance, $color) = @{$cref};
	}

	return undef unless (defined($code));
	my ($type, $modifier) = split('\.', $code);
	$type = 0 unless ($type);
	$modifier = 0 unless ($modifier);
	my $ref = $code2str[$type]->[$modifier];
	return undef unless ($ref);
	return undef unless ($type == $ref->{type} && $modifier == $ref->{modifier});
	$ref->{chance} = $chance;
	$ref->{chance_color} = $color if ($color);
	return $ref;
}

sub code_to_string {
	my ($code, $day) = @_;

	my $data = code_to_data($code);
	if (defined($day) && $day > 0 && $code >= SKY_CLR) {
		my $tdata = code_to_data($code+0.3);
		$data = $tdata if ($tdata);
	}
	my $str = (defined($data)) ? __($data->{description}) : undef;
	return $str;
}

sub cld_token_to_code {
	my ($token) = @_;

	#FIXME: need a way to indicate presence of TCU|CB in strings
	return SKY_FEW if ($token =~ /-OVC/o);
	return SKY_OVC if ($token =~ /OVC/o);
	return SKY_FEW if ($token =~ /-BKN/o);
	return SKY_MCLDY if ($token =~ /BKN/o);
	return SKY_CLDY if ($token =~ /VV/o || $token =~ /(?:P|W)\d+X/);
	return SKY_SCT if ($token =~ /SCT/o);
	return SKY_MCLR if ($token =~ /FEW/o);
	return undef if ($token =~ /CLR BLO|M/o);
	return SKY_CLR if ($token =~ /CLR|SKC|NCD|NSC/o);
	return undef;
}

sub opacity_to_code {
	my ($opacity) = @_;

	return undef if ($opacity < 0);
	return SKY_CLR if ($opacity <= 1);

	return SKY_MCLDY if ($opacity <= 8);
	return SKY_CLDY if ($opacity == 9);
	return SKY_OVC if ($opacity == 10);
	return undef;
}

		SNRA => WX_RASN_AND, SNUP => WX_SNUP_AND, UPSN => WX_SNUP_AND,
		SG => WX_SG, IC => WX_IC, SC => WX_SC, PL => WX_PL, GR => WX_GR, GS => WX_GS,
		SHRA => WX_RASH, SHSN => WX_SNSH, SHPL => WX_PL, SHGS => WX_GS, SHGR => WX_GR,
		RASH => WX_RASH, SNSH => WX_SNSH, PLSH => WX_PL, GSSH => WX_GS, GRSH => WX_GR,
		SHIC => WX_IC, ICSH => WX_IC, SHSG => WX_SG, SGSH => WX_SG, SHDZ => WX_DZ, DZSH => WX_DZ,
		SHRASN => WX_RASNSH_AND, SHSNRA => WX_RASNSH_AND, SHUP => WX_UP, SH => WX_SH,
		TS => WX_TH, TSRA => WX_TSRA, TSSN => WX_TSSN, TSPL => WX_TSPL, TSGS => WX_TSGS,
		TSGR => WX_TSGR, TSUP => WX_TS, TSRASN => WX_TSRASN_AND, TSSNRA => WX_TSRASN_AND,
		LT => WX_LT, UP => WX_UP, PO => WX_PO, FC => WX_FC, DS => WX_DS, SS => WX_DS,
		BLDU => WX_BLDU, DRDU => WX_DRDU, BLSA => WX_BLDU, DRSA => WX_DRDU,
	);

	my %other_matches = (
			BR => WX_BR, FG => WX_FG, FZFG => WX_FZFG, FU => WX_FU, VA => WX_VA, DU => WX_DU,
			SA => WX_DU, HZ => WX_HZ, SQ => WX_SQ, MIFG => WX_MIFG, BCFG => WX_BCFG,
			PRFG => WX_PRFG,
		);

	my %matches = (
		SH => WX_SH, DZ => WX_DZ, FZDZ => WX_FZDZ, RA => WX_RA, SHRA => WX_RASH,
		FZRA => WX_FZRA, SN => WX_SN, BLSN => WX_BLSN, DRSN => WX_DRSN, SHSN => WX_SNSH,
		SG => WX_SG, IC => WX_IC, SC => WX_SC, PL => WX_PL, GR => WX_GR, GS => WX_GS,
		BR => WX_BR, FG => WX_FG, FZFG => WX_FZFG, FU => WX_FU, VA => WX_VA, DU => WX_DU,
		SA => WX_DU, BLDU => WX_BLDU, DRDU => WX_DRDU, BLSA => WX_BLDU, DRSA => WX_DRDU,
		HZ => WX_HZ, PO => WX_PO, SQ => WX_SQ, FC => WX_FC, DS => WX_DS, SS => WX_DS,
		TS => WX_TH, UP => WX_UP, LT => WX_LT, MIFG => WX_MIFG, BCFG => WX_BCFG,
		PRFG => WX_PRFG, TSRA => WX_TSRA, TSSN => WX_TSSN, TSPL => WX_TSPL, TSGS => WX_TSGS,
		TSGR => WX_TSGR, SHUP => WX_UP, TSUP => WX_TS, SHGS => WX_GS, SHGR => WX_GR,
		SHPL => WX_PL, SNUP => WX_SNUP_AND, UPSN => WX_SNUP_AND, DZRA => WX_RADZ_AND,
		RADZ => WX_RADZ_AND, RASN => WX_RASN_AND, SNRA => WX_RASN_AND, SHRASN => WX_RASNSH_AND,
		SHSNRA => WX_RASNSH_AND,
	);

	my %sao_matches = (
		TORNADO => WX_FC, FUNNEL => WX_FC, SQUALL => WX_SQ, LTNG => WX_LT,
		DFTS => WX_DRSN, IP => WX_PL, PE => WX_PL, PL => WX_PL, SP => WX_GS, IF => WX_FZFG,
		ZL => WX_FZDZ, ZR => WX_FZRA, BS => WX_BLSN, F => WX_FG, H => WX_HZ, K => WX_FU,
		L => WX_DZ, P => WX_UP, R => WX_RA, S => WX_SN, T => WX_TS, A => WX_GR, W => WX_SH,
	);

	my $prefix="";
	if ($sao) {
		my $match_str = join('|', sort({length($b) <=> length($a)} keys(%sao_matches)));
		my $sao_re = qr/(TORNADO|FUNNEL|SQUALL|LTNG|DFTS|
			IP|IC|PE|PL|SP|IF|ZL|ZR|BS|
			F|H|K|L|P|R|S|T|A|W)(-{1,2}|\+)?/x;
		while ($token =~ /$sao_re/go) {
			my $tok = $1; $prefix = $2 if ($2);
			my $intensity = 0;
			if ($prefix eq "-") {
				$intensity = 0.1;
			} elsif ($prefix eq "--") {
				$intensity = 0.5;
			} elsif ($prefix eq "+") {
				$intensity = 0.2;
			}
			next unless (exists($sao_matches{$tok}));
			push(@codes, $sao_matches{$tok}+$intensity);
		}
	} else {
		my $match_str = join('|', 
			sort({length($b) <=> length($a)} keys(%intensity_matches),  keys(%matches)));
		my $metar_re = qr/($match_str)/;
		if ($token =~ /^(\+|-|VC|RE)?(.*)/o) {
			$prefix = $1 if ($1);
			my $str = $2;
			my $intensity = 0;
			my $modifier = 0;
			if ($prefix eq "RE") {
				$modifier = 0.3;
			} elsif ($prefix eq "VC") {
				$modifier = 0.4;
			}
			if ($prefix eq "-") {
				$intensity = 0.1;
			} elsif ($prefix eq "--") {
				$intensity = 0.5;
			} elsif ($prefix eq "+") {
				$intensity = 0.2;
			} elsif ($prefix eq "RE") {
				$intensity = 0.3;
			} elsif ($prefix eq "VC") {
				$intensity = 0.4;
			}
			while ($str =~ /$metar_re/go) {
				my $tok = $1;
				if (exists($intensity_matches{$tok})) {
					push(@codes, $intensity_matches{$tok}+$intensity);
				} elsif (exists($other_matches{$tok})) {
					push(@codes, $other_matches{$tok}+$modifier);
				}
			}	
		}
	}

	return @codes;
}

sub plain_language_to_wrng_codes {
   my $str = shift;
   my @codes;
   my @sentences = split(/\./o, $str);

   for my $s (@sentences){
      for my $key (keys(%wrn_matches)){
         if (exists $wrn_matches{$key}->{match} and $s =~ $wrn_matches{$key}->{match}){
            if (exists $wrn_matches{$key}->{not_match} and $s =~ $wrn_matches{$key}->{not_match}){
               next;
            }

            push @codes, $wrn_matches{$key}->{code};
         }
      }
	}

   if (scalar @codes < 1){
      push @codes, WRN_DEFAULT;
   }

   return @codes;
}

sub plain_language_to_codes {
	my $str = shift;
	my $cld_code = undef;
	my @sentences = split(/\./o, $str);
	my @sky_codes = ();
	my @obst_codes = ();
	my @impact = (0, 0);
	my @non_impact = (0, 0);
	my @codes = ();

	sub combine_impact {
		my ($current, $new, %ip) = @_;
		return @{$new} unless (exists($impact_priority{$current->[0]}));
		return @{$current} unless (exists($impact_priority{$new->[0]}));
		my @sorted = sort({
			my $ac = $a->[0]; my $bc = $b->[0];
			if (int($ac) == int($bc)) {
				return $b->[1] <=> $a->[1];
			} else {
				return $impact_priority{$ac} <=> $impact_priority{$bc};
			}
		} $current, $new);
		return @{$sorted[0]};
	}

	sub combine_non_impact {
		my ($current, $new, %ip) = @_;
		return @{$new} unless (exists($non_impact_priority{$current->[0]}));
		return @{$current} unless (exists($non_impact_priority{$new->[0]}));
		my $ic = int($current->[0]); my $in = int($new->[0]);
		if (($current->[1] eq $new->[1])
			&& ((($ic == WX_RASH || $ic == WX_RA || $ic == WX_DZ || $ic == WX_SH) && ($in == WX_SNSH || $in == WX_SN))
			|| (($ic == WX_SNSH || $ic == WX_SN) && ($in == WX_RASH || $in == WX_RA || $in == WX_DZ || $in == WX_SH)))
		) { 
			return (WX_RASN_AND, $current->[1]);
		}
		my @sorted = sort({
			my $ac = $a->[0]; my $bc = $b->[0];
			if ($b->[1] eq $a->[1]) {
				return $non_impact_priority{$ac} <=> $non_impact_priority{$bc};
			} else {
				return $b->[1] <=> $a->[1];
			}
		} $current, $new);
		return @{$sorted[0]};
	}

	#if we are processing for a warning object only need to do things in this block. Not worried about rest
	if ($sentences[0] =~ qr/(warning entity:)/i){
      for my $s (@sentences){
         for my $key (keys(%wrn_matches)){
            if (exists $wrn_matches{$key}->{match} and $s =~ $wrn_matches{$key}->{match}){
               if (exists $wrn_matches{$key}->{not_match} and $s =~ $wrn_matches{$key}->{not_match}){
                  next;
               }

               push @codes, $wrn_matches{$key}->{code};
            }
         }
		}

      if (scalar @codes < 1){
         push @codes, WRN_DEFAULT;
      }

      return ($cld_code, @codes);
	}

	foreach my $s (@sentences) {
		if ($s =~ /$match_str/o) {
			$s = $1 if ($s =~ /(.*?)\s(?:EXCEPT|SAUF)/);
			my @phrases = split(/$split_re/, $s);
			foreach my $p (@phrases) {
				$p =~ s/$time_re//g;
				$p =~ s/^\s+|\s+$//g;
				$p =~ s/\s{2,}/ /g;
				next unless ($p);
				MetManager::trace($TRACEID,9,"process '$p'");
				if (exists($cloud_matches{$p})) {
					MetManager::trace($TRACEID,9,"cloud match");
					push(@sky_codes, $cloud_matches{$p});
				} elsif ($p =~ /$mix_re/) {
					MetManager::trace($TRACEID,9,"mix match");
					push(@sky_codes, SKY_OVC);
					my $chance = $2||100;
					my ($t1, $t2, $t3) = ($3||'', $4||'', $5||'');
					#print STDERR "$t1, $t2, $t3\n";
					if (exists($impact_matches{$t1})) {
						@impact = combine_impact(\@impact, [$impact_matches{$t1}, $chance]);
					} 
					if (exists($impact_matches{$t2})) {
						@impact = combine_impact(\@impact, [$impact_matches{$t2}, $chance]);
					}
					if (exists($impact_matches{$t3})) {
						@impact = combine_impact(\@impact, [$impact_matches{$t3}, $chance]);
					}
					if (exists($non_impact_matches{$t1})) {
						@non_impact = combine_non_impact(\@non_impact,
							[$non_impact_matches{$t1}, $chance]);
					}
					if (exists($non_impact_matches{$t2})) {
						@non_impact = combine_non_impact(\@non_impact,
							[$non_impact_matches{$t2}, $chance]);
					}
					if (exists($non_impact_matches{$t3})) {
						@non_impact = combine_non_impact(\@non_impact,
							[$non_impact_matches{$t3}, $chance]);
					}
					if (exists($obstruction_matches{$t1})) {
						push(@obst_codes, $obstruction_matches{$t1});
					}
					if (exists($obstruction_matches{$t2})) {
						push(@obst_codes, $obstruction_matches{$t2});
					}
					if (exists($obstruction_matches{$t3})) {
						push(@obst_codes, $obstruction_matches{$t3});
					}
				} elsif ($p =~ /$chance_impact_re/) {
					MetManager::trace($TRACEID,9,"chance of a impact match");
					push(@sky_codes, SKY_OVC);
					my $chance = $2||20;
					@impact = combine_impact(\@impact, [$impact_matches{$3}, $chance]);
					MetManager::trace($TRACEID,9,"new impact=".join("|",@impact));
				} elsif ($p =~ $chance_rasn_re) {
					my $code = WX_RASN_AND;
					$code += $hvy if ($p =~ /$hvy_re/);
					MetManager::trace($TRACEID,9,"chance of a rain/snow match");
					push(@sky_codes, SKY_OVC);
					my $chance = $2||0;
					@non_impact = combine_non_impact(\@non_impact, [$code, $chance]);
				} elsif (exists($impact_matches{$p})) {
					MetManager::trace($TRACEID,9,"'$p' is a impact match");
					push(@sky_codes, SKY_OVC);
					@impact = combine_impact(\@impact, [$impact_matches{$p}, 100]);
				} elsif ($p =~ $rasn_re) {
					MetManager::trace($TRACEID,9,"rain/snow match");
					my $code = WX_RASN_AND;
					$code += $hvy if ($p =~ /$hvy_re/);
					push(@sky_codes, SKY_OVC);
					@non_impact = combine_non_impact(\@non_impact, [$code, 100]);
				} elsif ($p =~ /$chance_non_impact_re/) {
					MetManager::trace($TRACEID,9,"chance of a non impact match");
					push(@sky_codes, SKY_OVC);
					my $chance = $2||0;
					@non_impact = combine_non_impact(\@non_impact,
						[$non_impact_matches{$3}, $chance]);
				} elsif (exists($non_impact_matches{$p})) {
					MetManager::trace($TRACEID,9,"non impact match");
					push(@sky_codes, SKY_OVC);
					@non_impact = combine_non_impact(\@non_impact,
						[$non_impact_matches{$p}, 100]);
				} elsif (exists($obstruction_matches{$p})) {
					MetManager::trace($TRACEID,9,"obstruction match");
					push(@obst_codes, $obstruction_matches{$p});
				} else {
					MetManager::trace($TRACEID,9,"ignore it");
					if ($p =~ /$match_str/o 
						&& $p !~ /$local_effects_re/
						&& $p !~ /$ignore_re/
						&& $p !~ /$location_re/
					) {
						MetManager::trace($TRACEID,1,"'$p' doesn't match anything");
					}
				}
			}
		}
	}

	if (@sky_codes) {
		if (@sky_codes > 2 || int($sky_codes[0]) == SKY_CLR) {
			$cld_code = int(max(@sky_codes));
		} elsif (@sky_codes == 2) {
			if ($sky_codes[0] > $sky_codes[1]) {
				if ($sky_codes[0] == SKY_INCR) {
					$cld_code = SKY_MIX
				} else {
					$cld_code = int($sky_codes[0])+0.1;
				}
			} elsif ($sky_codes[0] < $sky_codes[1]) {
				if ($sky_codes[0] == SKY_DECR) {
					$cld_code = SKY_MIX
				} else {
					$cld_code = int($sky_codes[0])+0.2;
				}
			} else {
				$cld_code = $sky_codes[0];
			}
		} else {
			$cld_code = $sky_codes[0];
		}
		my $icc = int($cld_code);
		$cld_code = $icc if ($icc == SKY_INCR || $icc == SKY_DECR);
	}

	if (@obst_codes) {
		@obst_codes = sort({
			return $obstruction_priority{$a} <=> $obstruction_priority{$b};
		} @obst_codes);
		push(@codes, $obst_codes[0]);
	}

	if ($impact[0]) {
		$impact[1] = N__("RISK") if ($impact[1] == 20);
		push(@codes, [@impact, 'red']);
	} elsif ($non_impact[0]) {
		push(@codes, [@non_impact, 'black']);
	}

	$cld_code = SKY_CLR unless ($cld_code || @codes);
	return ($cld_code, @codes);
}

sub get_codes_from_sas {
	my $data = shift;

	my $cldcode = undef;
	$cldcode = MetManager::conditions::opacity_to_code(0)
		if (exists($data->{has_nsc}) || exists($data->{has_cavok}));

	if (exists($data->{opacity})) {
		$cldcode = MetManager::conditions::opacity_to_code($data->{opacity});
	}

	if (exists($data->{sky})) {
		my $tcode = MetManager::conditions::cld_token_to_code($data->{sky});
		$cldcode = $tcode if (defined($tcode) && (!defined($cldcode) || $tcode > $cldcode));
	}

	my @wx_codes = ();
	my $is_sa = $data->{is_sa} || $data->{is_sp};
	if (exists($data->{weather})) {
		foreach my $tok (split(' ', $data->{weather})) {
			push(@wx_codes, MetManager::conditions::wx_token_to_code($tok, $is_sa));
		}
	}

	my @re_wx_codes = ();
	if (exists($data->{recent_weather})) {
		foreach my $tok (split(' ', $data->{recent_weather})) {
			push(@re_wx_codes, MetManager::conditions::wx_token_to_code($tok, $is_sa));
		}
	}

	my $present = [@wx_codes] if (@wx_codes);
	my $recent = [@re_wx_codes] if (@re_wx_codes);
	return ($cldcode, $present, $recent);
}

sub get_codes_from_sms {
	my $sm = shift;
	my $present_wx_table = 4677;
	my $past_wx_table = 4561;
	my $cldcode = undef;
	my @wx_codes = ();
	my @re_wx_codes = ();
	my $ix = $sm->{s111}->{ix};
	my $cld_n = $sm->{s111}->{cld_n};

	return () if ((!defined($ix) || $ix == 3 || $ix == 6) && !defined($cld_n));

	if (defined($ix)) {
		if ($ix >= 4) {
			$present_wx_table = 4680;
			$past_wx_table = 4531;
		}

		my $ww = $sm->{s111}->{ww};
		my $ww1 = $sm->{s111}->{ww1};
		my $ww2 = $sm->{s111}->{ww2};
		if (defined($ww) && $ww > 3) {
			my @codes = MetManager::conditions::wmo_table_to_code($ww, $present_wx_table);
			foreach my $c (@codes) {
				next unless ($c);
				if (($present_wx_table == 4677 && ($ww >= 20 && $ww < 30)) ||
					($present_wx_table == 4680 && ($ww>= 20 && $ww < 27))
				) {
					push(@re_wx_codes, $c);
				} else {
					push(@wx_codes, $c);
				}
			}
		}
		unless (@re_wx_codes) {
			if (defined($ww1) && $ww1 > 2) {
				my @codes = MetManager::conditions::wmo_table_to_code($ww1, $past_wx_table);
				foreach my $c (@codes) {
					next unless ($c);
					push(@re_wx_codes, $c);
				}
			}
			if (defined($ww2) && $ww2 > 2) {
				if (defined($ww1) && $ww1 eq $ww2) {
					#no need to push the same code twice
				} else {
					my @codes = MetManager::conditions::wmo_table_to_code($ww2, $past_wx_table);
					foreach my $c (@codes) {
						next unless ($c);
						push(@re_wx_codes, $c);
					}
				}
			}
		}
	}

	if (defined($cld_n)) {
		$cldcode = MetManager::conditions::wmo_table_to_code($cld_n, 2700);
	}


	my $present = [@wx_codes] if (@wx_codes);
	my $recent = [@re_wx_codes] if (@re_wx_codes);
	return ($cldcode, $present, $recent);
}

sub build_timeframe_string {
	my (@timeframes) = @_;
	my $str = '';
	if (@timeframes == 1) {
		my $tf = $timeframes[0];
		return build_string($tf->{daylight}, $tf->{cldcode}, $tf->{weather});
	} else {
	}
}

sub build_string {
	my ($daylight, $cldcode, $present_wx, $recent_wx) = @_;
	my @present_codes = @{$present_wx} if defined($present_wx);
	my @recent_codes = @{$recent_wx} if defined($recent_wx);

	return '' unless(defined($cldcode) || @present_codes);

	# Make a string using cloud code and present weather info
	my $cld_str = "";
	$cld_str = MetManager::conditions::code_to_string($cldcode, $daylight)
		if (defined($cldcode));

	my @strs = ();
	foreach my $c (@present_codes) {
		push(@strs, MetManager::conditions::code_to_string($c, $daylight))
			if (defined($c));
	}

	if (@strs) {
		return (join(", ", @strs));
	}
	# Can't use present weather info, use cloud code and recent weather info instead
	push @strs, $cld_str;
	foreach my $c (@recent_codes) {
		push(@strs, MetManager::conditions::code_to_string($c, $daylight))
			if (defined($c));
	}
	@strs = grep {defined $_ and $_ ne ''} @strs;	
	my $string = join(",", @strs);	
	return $string;
	
}

sub create_icon {
	my ($daylight, $cldcode, $present_wx, $recent_wx, $bg, $scale_factor) = @_;
	my @wx_codes = @{$present_wx} if defined($present_wx);

	my $icon_dir = MetManager::dir('data')."/images/conditions/";
	my $image = Graphics::Magick->new();
	if ($cldcode) {
		my $data = MetManager::conditions::code_to_data($cldcode);
		$daylight = 1 if ($data->{modifier} == 3);
	}

	my ($nightbg, $daybg);
	if (not $bg) {
		($bg, $nightbg, $daybg) = (undef, undef, undef);
	} elsif (ref($bg) eq 'ARRAY') {
		($nightbg, $daybg) = @{$bg};
	} else {
		($nightbg, $daybg) = (qw/223366 BBCCFF/);
	}

	if (not defined $daylight) {
		$image->Read(filename=>$icon_dir."Nightime_bg.ico");
		$bg = $nightbg if defined $nightbg;
	} elsif (ref($daylight) eq 'ARRAY') {
		if ($daylight->[0] < 1) {
			if ($daylight->[1] > 0) {
				$image->Read(filename=>$icon_dir."Night_to_Day.ico");
			} else {
				$image->Read(filename=>$icon_dir."Nightime_bg.ico");
			}
			$bg = $nightbg if defined $nightbg;
		} else {
			if ($daylight->[-1] < 0) {
				$image->Read(filename=>$icon_dir."Day_to_Night.ico");
			} else {
				$image->Read(filename=>$icon_dir."Daytime_bg.ico");
			}
			$bg = $daybg if defined $daybg;
		}
	} elsif ($daylight < 1) {
		$image->Read(filename=>$icon_dir."Nightime_bg.ico");
		$bg = $nightbg if defined $nightbg;
	} else {
		$image->Read(filename=>$icon_dir."Daytime_bg.ico");
		$bg = $daybg if defined $daybg;
	}

	if (defined $bg) {
		my $bg_image = Graphics::Magick->new(
			size => $image->Get('width') . 'x' . $image->Get('height'));
		$bg_image->Read('xc:#'.$bg);
		$bg_image->Composite(compose => 'Over', image => $image);
		$image->Composite(compose => 'Copy', image => $bg_image);
	}

	my $has_freezing=0;
	my $need_cloud=0;
	my $need_lt=0;
	my $convective=0;
	my $heavy=0;
	my $obscuration_icon=undef;
	my $need_icon=1;
	my $state=0;
	my $ctxt = '';
	my $color = 'white';
	foreach my $c (@wx_codes) {
		next unless (defined($c));
		my $data = MetManager::conditions::code_to_data($c);
		next unless ($data);
		$heavy=1 if ($data->{modifier} == 2 && $data->{flag} & PRECIPITATION);
		$has_freezing=1 if ($data->{flag} & FREEZING);
		$need_cloud=1 if ($data->{flag} & PRECIPITATION);
		$convective=1 if ($data->{flag} & CONVECTIVE);
		$need_lt=1 if ($data->{flag} & LIGHTNING);
		if ($data->{flag} & OBSCURATION) {
			$obscuration_icon = $data->{icon} unless ($obscuration_icon);
			next;
		}
		if ($data->{type} == WX_BLZD) {
			$obscuration_icon = "Pcpn_blowing_snow_heavy";
			$heavy=$need_cloud=1;
		}
		my $timage = Graphics::Magick->new();
		if ($data->{icon} && $need_icon) {
			$need_icon = 1;
			my $rc = $timage->Read(filename=>$icon_dir.$data->{icon}.'.ico');
			$image->Composite(image=>$timage, compose => 'Over') unless $rc;
		}
		if ($data->{chance} eq "RISK") {
			$ctxt = $data->{chance};
			$color = $data->{chance_color} if ($data->{chance_color});
		} elsif ($data->{chance} < 100) {
			$ctxt = sprintf(" %d%% ", $data->{chance});
			$color = $data->{chance_color} if ($data->{chance_color});
		}
	}

	if ($need_lt) {
		my $iconfile = $icon_dir."lightning.ico";
		$iconfile = $icon_dir."frequent_lightning.ico" if ($heavy);
		my $timage = Graphics::Magick->new();
		my $rc = $timage->Read(filename=>$iconfile);
		$image->Composite(image=>$timage, compose => 'Over') unless $rc;
	}

	if ($cldcode) {
		my $data = MetManager::conditions::code_to_data($cldcode);
		$state=$data->{flag} if ($data->{flag} & (INCREASING|DECREASING));
		if ($data && $data->{icon}) {
			my $icon = $data->{icon};
			if ($convective) {
				$icon =~ s/Clouds_/Clouds_cvctv_/og;
			} elsif ($heavy) {
				$icon = 'Clouds_ovc_with_hvy_pcpn';
			}
			my $timage = Graphics::Magick->new();
			my $rc = $timage->Read(filename=>$icon_dir.$icon.'.ico');
			$image->Composite(image=>$timage, compose => 'Over') unless $rc;
		}
	} elsif ($need_cloud || $convective) {
		my $icon = "Clouds_bkn.ico";
		$icon = "Clouds_cvctv_bkn.ico" if ($convective);
		$icon = "Clouds_cvctv_ovc.ico" if ($heavy);
		my $timage = Graphics::Magick->new();
		my $rc = $timage->Read(filename=>$icon_dir.$icon);
		$image->Composite(image=>$timage, compose => 'Over') unless $rc;
	}

	if ($obscuration_icon) {
		my $timage = Graphics::Magick->new();
		my $rc = $timage->Read(filename=>$icon_dir.$obscuration_icon.'.ico');
		$image->Composite(image=>$timage, compose => 'Over') unless $rc;
	}

	if ($has_freezing) {
		my $timage = Graphics::Magick->new();
		my $rc = $timage->Read(filename=>$icon_dir."freezing.ico");
		$image->Composite(image=>$timage, compose => 'Over') unless $rc;
	}
	unless ($ctxt) {
		if ($state & INCREASING) {
			my $timage = Graphics::Magick->new();
			my $rc = $timage->Read(filename=>$icon_dir."increasing.ico");
			$image->Composite(image=>$timage, compose => 'Over') unless $rc;
		} elsif ($state & DECREASING) {
			my $timage = Graphics::Magick->new();
			my $rc = $timage->Read(filename=>$icon_dir."decreasing.ico");
			$image->Composite(image=>$timage, compose => 'Over') unless $rc;
		}
	}

	my $ps = 27;
	my $xoffset = 20;
	my $yoffset = 0;
	if ($scale_factor) {
		$ps*=$scale_factor;
		$xoffset*=$scale_factor;
		my ($w, $h) = $image->Get('width', 'height');
		$image->Resize(width => $w*$scale_factor, height => $h*$scale_factor);
	}
	$image->Annotate(fill=>$color, gravity=>'center', pointsize=>$ps,
		text => __($ctxt), geometry => "-$xoffset+$yoffset") if ($ctxt);
	return $image;
}

1;

=head1 NAME

MetManager::conditons - Perl module for mapping of weather codes (METAR, synoptic, etc)
into more plain language strings and icons

=head1 DESCRIPTION

=head2 FUNCTIONS

=over

=item code_to_string($code, $day)

Takes a single code and an optional day indicator and returns the appropriate string.
If the 'day' variable is passed and is greater than zero then it is considered daytime,
otherwise nighttime is assumed

=item cld_token_to_code($token)

Takes a METAR style cloud description string (FEW, BKN, etc) and returns code which
reflects the worst occurence. ie if the string was 'FEW020 BKN100', it would return the code
for broken sky.  Returns undef if no match found.

=item opacity_to_code($token)

Takes a value between 0 and 10 (usually the opacity field from a METAR report) and returns a
code.  Returns undef if value is not between 0 and 10.

=item wmo_table_to_code($wmo_code, $table)

Takes a WMO code value and a WMO code table and returns the appropriate code.  Returns undef
if the WMO code is unused or doesn't exist.  It works on the following WMO code tables: 
2700, 4561, 4531, 4677, 4680

=item wx_token_to_code($token, $sao)

Takes a METAR weather token and returns a list of the appropriate codes.  If the $sao flag
is set, then it will work on the old SAO weather tokens.

=item plain_language_to_codes($string)

Takes a plain language string (like from a public forecast) and returns codes describing the situation.  First element in the returned list is the cloud/sky code (could be undef) while the rest are weather codes

=item get_codes_from_sas($data)

Takes a reference to a SAS data structure and returns an array with the following elements:
a code indicating the sky condtion, a reference to an array of the codes representing the
present weather, a reference to an array of the codes representing the past weather if any.
The elements from the SAS data structure that are used are: has_nsc, opacity, sky, is_sa,
is_sp, weather, recent_weather

=item get_codes_from_sms($data)

Takes a reference to a SMS data structure and returns an array with the following elements:
a code indicating the sky condtion, a reference to an array of the codes representing the
present weather, a reference to an array of the codes representing the past weather if any.
The elements from the SMS data structure that are used are: s111->{ix}, s111->{cld_n},
s111->{ww}, s111->{ww}, s111->{ww}

=item build_string($daylight, $cldcode, $present_wx, $recent_wx)

Takes a day indicator ($day < 0 equals night, $day > 0 equals day, 0 indicates unknown which
currently gets treated as nighttime), the code describing the current sky conditions,
a reference to an array containing all the present weather codes, a reference to an array
containing all the recent weather codes.  It returns a string describing the situation.

=item create_icon($day, $cldcode, $present_wx, $recent_wx, $bg)

Takes a day indicator ($day == undef or <= 0 equals night, $day > 0 equals day, 0 indicates unknown which
currently gets treated as nighttime); the code describing the current sky conditions;
a reference to an array containing all the present weather codes; a reference to an array
containing all the recent weather codes; and $bg which can be specified as a scalar or array-reference. 
if specified as a scalar then undefined/zero means no-background and non-zero means do a background 
using API defaults. If specified as an array reference then the caller has the option to
pass the [nightbg, daybg] and each are specified as hexidecimal r/g/b string 'RRGGBB'. ie) '33CCFF'.
The API returns a Graphics::Magick image object of the built icon that best represents the situation
described by the values entered.

=back
